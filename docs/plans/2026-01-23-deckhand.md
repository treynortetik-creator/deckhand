**Goal:** Build an AI-powered slide deck generator that outputs to Google Slides and PPTX, using OpenRouter for LLM and image generation, with a pirate-themed premium UI.

**Architecture:**
- Frontend: React + TypeScript + Tailwind CSS + Lucide icons
- Backend: Python FastAPI with async support for parallel generation

```python
import asyncio
from logging.config import fileConfig

from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config

from alembic import context

from app.config import get_settings
from app.database import Base
from app.models import *  # noqa: F401, F403

config = context.config
settings = get_settings()

config.set_main_option("sqlalchemy.url", settings.database_url)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)

    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

```python
from datetime import datetime, timedelta
from typing import Optional

from jose import JWTError, jwt
from passlib.context import CryptContext
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.config import get_settings
from app.models.user import User

settings = get_settings()
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)


def get_password_hash(password: str) -> str:
    return pwd_context.hash(password)


def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.access_token_expire_minutes)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.secret_key, algorithm=settings.algorithm)
    return encoded_jwt


def decode_token(token: str) -> Optional[str]:
    try:
        payload = jwt.decode(token, settings.secret_key, algorithms=[settings.algorithm])
        email: str = payload.get("sub")
        return email
    except JWTError:
        return None


async def get_user_by_email(db: AsyncSession, email: str) -> Optional[User]:
    result = await db.execute(select(User).where(User.email == email))
    return result.scalar_one_or_none()


async def create_user(db: AsyncSession, email: str, password: str, full_name: str = None) -> User:
    hashed_password = get_password_hash(password)
    user = User(email=email, hashed_password=hashed_password, full_name=full_name)
    db.add(user)
    await db.commit()
    await db.refresh(user)
    return user


async def authenticate_user(db: AsyncSession, email: str, password: str) -> Optional[User]:
    user = await get_user_by_email(db, email)
    if not user:
        return None
    if not verify_password(password, user.hashed_password):
        return None
    return user
```

```python
import re
from typing import Optional
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from PyPDF2 import PdfReader
import io

from app.models.brand import Brand


# Common color name to hex mapping
COLOR_MAP = {
    "red": "#FF0000", "blue": "#0000FF", "green": "#00FF00",
    "yellow": "#FFFF00", "orange": "#FFA500", "purple": "#800080",
    "black": "#000000", "white": "#FFFFFF", "gray": "#808080",
    "navy": "#000080", "teal": "#008080",
}


def extract_hex_colors(text: str) -> list[str]:
    """Extract hex color codes from text."""
    pattern = r'#(?:[0-9a-fA-F]{3}){1,2}\b'
    return list(set(re.findall(pattern, text)))


def extract_rgb_colors(text: str) -> list[str]:
    """Extract RGB colors and convert to hex."""
    pattern = r'rgb\s*\(\s*(\d{1,3})\s*,\s*(\d{1,3})\s*,\s*(\d{1,3})\s*\)'
    matches = re.findall(pattern, text, re.IGNORECASE)
    hex_colors = []
    for r, g, b in matches:
        hex_color = f"#{int(r):02x}{int(g):02x}{int(b):02x}"
        hex_colors.append(hex_color)
    return hex_colors


def extract_fonts(text: str) -> list[str]:
    """Extract font family names from text."""
    common_fonts = [
        "Arial", "Helvetica", "Times New Roman", "Georgia", "Verdana",
        "Roboto", "Open Sans", "Lato", "Montserrat", "Poppins",
        "Inter", "Playfair Display", "Source Sans Pro", "Raleway",
    ]
    found = []
    text_lower = text.lower()
    for font in common_fonts:
        if font.lower() in text_lower:
            found.append(font)
    return found


async def extract_brand_from_pdf(pdf_content: bytes) -> dict:
    """Extract brand guidelines from PDF content."""
    reader = PdfReader(io.BytesIO(pdf_content))
    full_text = ""
    for page in reader.pages:
        full_text += page.extract_text() or ""

    # Extract colors
    hex_colors = extract_hex_colors(full_text)
    rgb_colors = extract_rgb_colors(full_text)
    all_colors = list(set(hex_colors + rgb_colors))

    # Extract fonts
    fonts = extract_fonts(full_text)

    # Build response
    primary_colors = {}
    secondary_colors = {}

    if len(all_colors) >= 1:
        primary_colors["primary"] = all_colors[0]
    if len(all_colors) >= 2:
        primary_colors["secondary"] = all_colors[1]
    if len(all_colors) >= 3:
        secondary_colors["accent"] = all_colors[2]
    for i, color in enumerate(all_colors[3:], start=1):
        secondary_colors[f"color_{i}"] = color

    font_dict = {}
    if len(fonts) >= 1:
        font_dict["heading"] = fonts[0]
    if len(fonts) >= 2:
        font_dict["body"] = fonts[1]

    return {
        "primary_colors": primary_colors,
        "secondary_colors": secondary_colors,
        "fonts": font_dict,
        "guidelines_text": full_text[:5000],  # Limit stored text
    }


async def get_brand(db: AsyncSession) -> Optional[Brand]:
    """Get the brand (singleton for MVP)."""
    result = await db.execute(select(Brand).limit(1))
    return result.scalar_one_or_none()


async def create_or_update_brand(db: AsyncSession, brand_data: dict) -> Brand:
    """Create or update the brand."""
    brand = await get_brand(db)
    if brand:
        for key, value in brand_data.items():
            if value is not None:
                setattr(brand, key, value)
    else:
        brand = Brand(**brand_data)
        db.add(brand)
    await db.commit()
    await db.refresh(brand)
    return brand
```

```python
import os
import uuid
from typing import Optional
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from fastapi import UploadFile

from app.config import get_settings
from app.models.asset import Asset

settings = get_settings()

ALLOWED_IMAGE_TYPES = {"image/png", "image/jpeg", "image/jpg", "image/svg+xml", "image/webp"}
ALLOWED_DOC_TYPES = {"application/pdf", "application/vnd.openxmlformats-officedocument.wordprocessingml.document"}
ALLOWED_TYPES = ALLOWED_IMAGE_TYPES | ALLOWED_DOC_TYPES


async def save_upload_file(file: UploadFile) -> tuple[str, int]:
    """Save uploaded file and return path and size."""
    os.makedirs(settings.upload_dir, exist_ok=True)

    ext = os.path.splitext(file.filename)[1] if file.filename else ""
    unique_name = f"{uuid.uuid4()}{ext}"
    file_path = os.path.join(settings.upload_dir, unique_name)

    content = await file.read()
    file_size = len(content)

    with open(file_path, "wb") as f:
        f.write(content)

    return file_path, file_size


async def delete_file(file_url: str) -> None:
    """Delete file from filesystem."""
    if os.path.exists(file_url):
        os.remove(file_url)


async def create_asset(
    db: AsyncSession,
    filename: str,
    file_type: str,
    file_url: str,
    file_size: int,
    tags: list[str],
    user_id: int,
) -> Asset:
    """Create new asset record."""
    asset = Asset(
        filename=filename,
        file_type=file_type,
        file_url=file_url,
        file_size=file_size,
        tags=tags,
        uploaded_by=user_id,
    )
    db.add(asset)
    await db.commit()
    await db.refresh(asset)
    return asset


async def get_asset(db: AsyncSession, asset_id: int) -> Optional[Asset]:
    """Get asset by ID."""
    result = await db.execute(select(Asset).where(Asset.id == asset_id))
    return result.scalar_one_or_none()


async def list_assets(
    db: AsyncSession,
    skip: int = 0,
    limit: int = 50,
    tag_filter: Optional[str] = None,
    type_filter: Optional[str] = None,
) -> tuple[list[Asset], int]:
    """List assets with optional filters."""
    query = select(Asset)
    count_query = select(func.count(Asset.id))

    if type_filter:
        if type_filter == "image":
            query = query.where(Asset.file_type.in_(ALLOWED_IMAGE_TYPES))
            count_query = count_query.where(Asset.file_type.in_(ALLOWED_IMAGE_TYPES))
        elif type_filter == "document":
            query = query.where(Asset.file_type.in_(ALLOWED_DOC_TYPES))
            count_query = count_query.where(Asset.file_type.in_(ALLOWED_DOC_TYPES))

    # Note: JSON array contains is DB-specific, simplified here
    query = query.order_by(Asset.uploaded_at.desc()).offset(skip).limit(limit)

    result = await db.execute(query)
    assets = result.scalars().all()

    count_result = await db.execute(count_query)
    total = count_result.scalar() or 0

    return list(assets), total


async def update_asset_tags(db: AsyncSession, asset_id: int, tags: list[str]) -> Optional[Asset]:
    """Update asset tags."""
    asset = await get_asset(db, asset_id)
    if asset:
        asset.tags = tags
        await db.commit()
        await db.refresh(asset)
    return asset


async def delete_asset(db: AsyncSession, asset_id: int) -> bool:
    """Delete asset and its file."""
    asset = await get_asset(db, asset_id)
    if not asset:
        return False

    await delete_file(asset.file_url)
    await db.delete(asset)
    await db.commit()
    return True
```

import httpx
import asyncio
from typing import Optional

from app.config import get_settings
from app.schemas.generation import SlideContent, DeckOutline

settings = get_settings()


class OpenRouterClient:
    def __init__(self):
        self.base_url = settings.openrouter_base_url
        self.api_key = settings.openrouter_api_key
        self.headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json",
            "HTTP-Referer": "https://deckhand.safelyyou.com",
            "X-Title": "Deckhand",
        }

    async def chat_completion(
        self,
        messages: list[dict],
        model: str = None,
        temperature: float = 0.7,
        max_tokens: int = 4000,
    ) -> str:
        """Send chat completion request to OpenRouter."""
        model = model or settings.default_llm_model

        async with httpx.AsyncClient(timeout=120.0) as client:
            response = await client.post(
                f"{self.base_url}/chat/completions",
                headers=self.headers,
                json={
                    "model": model,
                    "messages": messages,
                    "temperature": temperature,
                    "max_tokens": max_tokens,
                },
            )
            response.raise_for_status()
            data = response.json()
            return data["choices"][0]["message"]["content"]

    async def generate_image(
        self,
        prompt: str,
        model: str = None,
    ) -> Optional[str]:
        """Generate image via OpenRouter (returns URL or base64)."""
        model = model or settings.default_image_model

        try:
            async with httpx.AsyncClient(timeout=120.0) as client:
                response = await client.post(
                    f"{self.base_url}/images/generations",
                    headers=self.headers,
                    json={
                        "model": model,
                        "prompt": prompt,
                        "n": 1,
                        "size": "1024x1024",
                    },
                )
                response.raise_for_status()
                data = response.json()
                return data["data"][0].get("url") or data["data"][0].get("b64_json")
        except Exception as e:
            print(f"Image generation failed: {e}")
            return None


async def create_deck_outline(
    client: OpenRouterClient,
    prompt: str,
    template_structure: list[dict] | None,
    slide_count: int,
    tone: str,
    brand_context: str = "",
) -> DeckOutline:
    """Use orchestration agent to create deck outline."""

    template_context = ""
    if template_structure:
        template_context = f"""
Use this slide structure:
{template_structure}
"""

    system_prompt = f"""You are the Deckhand Orchestration Agent. Create a detailed outline for a presentation deck.

Brand context: {brand_context}

Requirements:
- Create exactly {slide_count} slides
- Tone: {tone}
- Each slide needs: title, type, body content OR bullets, and an image prompt if visual is needed
{template_context}

Output as JSON with this structure:
{{
  "title": "Deck Title",
  "summary": "Brief deck summary",
  "slides": [
    {{
      "slide_number": 1,
      "slide_type": "title",
      "title": "Slide Title",
      "body": "Main content or null",
      "bullets": ["point 1", "point 2"] or null,
      "image_prompt": "Detailed prompt for AI image generation" or null,
      "speaker_notes": "Optional notes"
    }}
  ]
}}
"""

    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": f"Create a presentation about: {prompt}"},
    ]

    response = await client.chat_completion(messages, temperature=0.7)

    # Parse JSON from response
    import json
    try:
        # Handle markdown code blocks
        if "```json" in response:
            response = response.split("```json")[1].split("```")[0]
        elif "```" in response:
            response = response.split("```")[1].split("```")[0]

        data = json.loads(response.strip())
        slides = [SlideContent(**s) for s in data["slides"]]
        return DeckOutline(
            title=data["title"],
            slides=slides,
            summary=data.get("summary", ""),
        )
    except json.JSONDecodeError as e:
        raise ValueError(f"Failed to parse outline JSON: {e}")


async def generate_slide_content(
    client: OpenRouterClient,
    slide: SlideContent,
    deck_context: str,
    brand_context: str,
) -> SlideContent:
    """Content agent: Generate detailed content for a single slide."""

    system_prompt = f"""You are a Deckhand Content Agent. Enhance and expand the slide content.

Deck context: {deck_context}
Brand context: {brand_context}

Improve the slide content with:
- Compelling, concise copy
- Clear messaging
- Professional tone
- If bullets, keep to 3-5 points max

Return enhanced slide as JSON matching the input structure.
"""

    messages = [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": f"Enhance this slide: {slide.model_dump_json()}"},
    ]

    response = await client.chat_completion(messages, temperature=0.6, max_tokens=1000)

    import json
    try:
        if "```json" in response:
            response = response.split("```json")[1].split("```")[0]
        elif "```" in response:
            response = response.split("```")[1].split("```")[0]

        data = json.loads(response.strip())
        return SlideContent(**data)
    except:
        return slide  # Return original if parsing fails


async def generate_slides_parallel(
    client: OpenRouterClient,
    outline: DeckOutline,
    brand_context: str,
    max_concurrent: int = 3,
) -> list[SlideContent]:
    """Generate all slide content in parallel batches."""

    deck_context = f"Title: {outline.title}\nSummary: {outline.summary}"

    async def process_slide(slide: SlideContent) -> SlideContent:
        enhanced = await generate_slide_content(client, slide, deck_context, brand_context)

        # Generate image if needed
        if enhanced.image_prompt:
            image_url = await client.generate_image(enhanced.image_prompt)
            enhanced.image_url = image_url

        return enhanced

    # Process in batches to respect rate limits
    results = []
    for i in range(0, len(outline.slides), max_concurrent):
        batch = outline.slides[i:i + max_concurrent]
        batch_results = await asyncio.gather(*[process_slide(s) for s in batch])
        results.extend(batch_results)

    return results


# Singleton client
_client: Optional[OpenRouterClient] = None

def get_openrouter_client() -> OpenRouterClient:
    global _client
    if _client is None:
        _client = OpenRouterClient()
    return _client

```python
async def generate_deck(
    db: AsyncSession,
    request: GenerationRequest,
    user_id: int,
    progress_callback: Optional[Callable] = None,
) -> GenerationResult:
    """
    Main deck generation orchestration.

    1. Get template and brand context
    2. Create deck outline (orchestration agent)
    3. Generate slide content in parallel (content agents)
    4. Generate images in parallel
    5. Create deck record
    """
    start_time = time.time()
    client = get_openrouter_client()

    # Progress helper
    def update_progress(status: str, step: int, total: int, message: str):
        if progress_callback:
            progress_callback(status, step, total, message)

    update_progress("started", 0, 5, "Initializing deck generation...")

    # Get template structure if specified
    template_structure = None
    if request.template_id:
        template = await get_template(db, request.template_id)
        if template:
            template_structure = template.slide_structure

    # Get brand context
    brand_context = ""
    brand = await get_brand(db)
    if brand:
        brand_context = f"""
Brand: {brand.name}
Colors: Primary {brand.primary_colors}, Secondary {brand.secondary_colors}
Fonts: {brand.fonts}
Guidelines: {brand.guidelines_text[:500] if brand.guidelines_text else 'None'}
"""

    update_progress("outlining", 1, 5, "Creating deck outline...")

    # Step 1: Create outline
    try:
        outline = await create_deck_outline(
            client=client,
            prompt=request.prompt,
            template_structure=template_structure,
            slide_count=request.slide_count,
            tone=request.tone,
            brand_context=brand_context,
        )
    except Exception as e:
        # Log failure
        history = GenerationHistory(
            prompt=request.prompt,
            template_id=request.template_id,
            assets_used=request.asset_ids,
            model_used=client.headers.get("model", "unknown"),
            success=False,
            error_message=str(e),
        )
        db.add(history)
        await db.commit()

    enhanced_slides = await generate_slides_parallel(
```

```python
def create_title_slide(prs: Presentation, slide: SlideContent, brand_colors: dict):
    """Create a title slide."""
    layout = prs.slide_layouts[6]  # Blank
    pptx_slide = prs.slides.add_slide(layout)

    # Title
    title_box = pptx_slide.shapes.add_textbox(
        Inches(0.5), Inches(2.5), Inches(9), Inches(1.5)
    )
    title_frame = title_box.text_frame
    title_para = title_frame.paragraphs[0]
    title_para.text = slide.title
    title_para.font.size = Pt(44)
    title_para.font.bold = True
    title_para.alignment = PP_ALIGN.CENTER

    if brand_colors.get("primary"):
        r, g, b = hex_to_rgb(brand_colors["primary"])
        title_para.font.color.rgb = RgbColor(r, g, b)

    # Subtitle/body
    if slide.body:
        sub_box = pptx_slide.shapes.add_textbox(
            Inches(0.5), Inches(4), Inches(9), Inches(1)
        )
        sub_frame = sub_box.text_frame
        sub_para = sub_frame.paragraphs[0]
        sub_para.text = slide.body
        sub_para.font.size = Pt(24)
        sub_para.alignment = PP_ALIGN.CENTER


def create_content_slide(prs: Presentation, slide: SlideContent, brand_colors: dict):
    """Create a content slide with title and body/bullets."""
    layout = prs.slide_layouts[6]  # Blank
    pptx_slide = prs.slides.add_slide(layout)

    # Title
    title_box = pptx_slide.shapes.add_textbox(
        Inches(0.5), Inches(0.5), Inches(9), Inches(1)
    )
    title_frame = title_box.text_frame
    title_para = title_frame.paragraphs[0]
    title_para.text = slide.title
    title_para.font.size = Pt(32)
    title_para.font.bold = True

    if brand_colors.get("primary"):
        r, g, b = hex_to_rgb(brand_colors["primary"])
        title_para.font.color.rgb = RgbColor(r, g, b)

    # Body or bullets
    content_top = Inches(1.7)
    content_height = Inches(4.5)

    if slide.image_url:
        # With image: content on left, image on right
        content_box = pptx_slide.shapes.add_textbox(
            Inches(0.5), content_top, Inches(4.5), content_height
        )
    else:
        # No image: full width content
        content_box = pptx_slide.shapes.add_textbox(
            Inches(0.5), content_top, Inches(9), content_height
        )

    content_frame = content_box.text_frame
    content_frame.word_wrap = True

    if slide.bullets:
        for i, bullet in enumerate(slide.bullets):
            if i == 0:
                p = content_frame.paragraphs[0]
            else:
                p = content_frame.add_paragraph()
            p.text = f"• {bullet}"
            p.font.size = Pt(18)
            p.space_after = Pt(12)
    elif slide.body:
        p = content_frame.paragraphs[0]
        p.text = slide.body
        p.font.size = Pt(18)


async def create_pptx(
    slides: list[SlideContent],
    title: str,
    brand_colors: dict = None,
    output_path: str = None,
) -> str:
    """Create PPTX file from slides."""
    prs = Presentation()
    prs.slide_width = Inches(10)
    prs.slide_height = Inches(7.5)

    brand_colors = brand_colors or {}

    for slide in slides:
        if slide.slide_type == "title":
            create_title_slide(prs, slide, brand_colors)
        else:
            create_content_slide(prs, slide, brand_colors)

    # Save
    os.makedirs(settings.upload_dir, exist_ok=True)
    if not output_path:
        safe_title = "".join(c for c in title if c.isalnum() or c in " -_")[:50]
        output_path = os.path.join(settings.upload_dir, f"{safe_title}.pptx")

    prs.save(output_path)
    return output_path
```

```python
def get_slides_service():
    """Get authenticated Google Slides API service."""
    if not settings.google_credentials_json:
        raise ValueError("Google credentials not configured")

    # Decode base64 credentials
    creds_json = base64.b64decode(settings.google_credentials_json).decode()
    creds_dict = json.loads(creds_json)

    credentials = service_account.Credentials.from_service_account_info(
        creds_dict,
        scopes=[
            'https://www.googleapis.com/auth/presentations',
            'https://www.googleapis.com/auth/drive',
        ]
    )

    return build('slides', 'v1', credentials=credentials)


def hex_to_rgb_float(hex_color: str) -> dict:
    """Convert hex color to Google Slides RGB format (0-1 floats)."""
    hex_color = hex_color.lstrip('#')
    r = int(hex_color[0:2], 16) / 255
    g = int(hex_color[2:4], 16) / 255
    b = int(hex_color[4:6], 16) / 255
    return {"red": r, "green": g, "blue": b}


async def create_google_slides_presentation(
    slides: list[SlideContent],
    title: str,
    brand_colors: dict = None,
) -> str:
    """Create a Google Slides presentation and return the URL."""
    service = get_slides_service()

    # Create presentation
    presentation = service.presentations().create(body={"title": title}).execute()
    presentation_id = presentation['presentationId']

    # Get the default slide ID to delete it later
    default_slide_id = presentation['slides'][0]['objectId']

    brand_colors = brand_colors or {}
    requests = []

    # Create slides
    for i, slide in enumerate(slides):
        slide_id = f"slide_{i}"
        title_id = f"title_{i}"
        body_id = f"body_{i}"

        # Create slide
        requests.append({
            "createSlide": {
                "objectId": slide_id,
                "insertionIndex": i,
                "slideLayoutReference": {"predefinedLayout": "BLANK"},
            }
        })

        # Add title
        requests.append({
            "createShape": {
                "objectId": title_id,
                "shapeType": "TEXT_BOX",
                "elementProperties": {
                    "pageObjectId": slide_id,
                    "size": {
                        "width": {"magnitude": 600, "unit": "PT"},
                        "height": {"magnitude": 50, "unit": "PT"},
                    },
                    "transform": {
                        "scaleX": 1, "scaleY": 1,
                        "translateX": 36, "translateY": 30,
                        "unit": "PT",
                    },
                },
            }
        })

        requests.append({
            "insertText": {
                "objectId": title_id,
                "text": slide.title,
            }
        })

        # Style title
        requests.append({
            "updateTextStyle": {
                "objectId": title_id,
                "style": {
                    "fontSize": {"magnitude": 28 if slide.slide_type == "title" else 24, "unit": "PT"},
                    "bold": True,
                },
                "fields": "fontSize,bold",
            }
        })

        # Add body content
        content_text = ""
        if slide.bullets:
            content_text = "\n".join(f"• {b}" for b in slide.bullets)
        elif slide.body:
            content_text = slide.body

        if content_text:
            requests.append({
                "createShape": {
                    "objectId": body_id,
                    "shapeType": "TEXT_BOX",
                    "elementProperties": {
                        "pageObjectId": slide_id,
                        "size": {
                            "width": {"magnitude": 600, "unit": "PT"},
                            "height": {"magnitude": 300, "unit": "PT"},
                        },
                        "transform": {
                            "scaleX": 1, "scaleY": 1,
                            "translateX": 36, "translateY": 100,
                            "unit": "PT",
                        },
                    },
                }
            })

            requests.append({
                "insertText": {
                    "objectId": body_id,
                    "text": content_text,
                }
            })

            requests.append({
                "updateTextStyle": {
                    "objectId": body_id,
                    "style": {
                        "fontSize": {"magnitude": 14, "unit": "PT"},
                    },
                    "fields": "fontSize",
                }
            })

    # Delete default blank slide
    requests.append({
        "deleteObject": {"objectId": default_slide_id}
    })

    # Execute all requests
    if requests:
        service.presentations().batchUpdate(
            presentationId=presentation_id,
            body={"requests": requests}
        ).execute()

    return f"https://docs.google.com/presentation/d/{presentation_id}/edit"
```

```python
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get all generated decks."""
    query = select(Deck).order_by(Deck.created_at.desc()).offset(skip).limit(limit)
    result = await db.execute(query)
    decks = result.scalars().all()

    count_result = await db.execute(select(func.count(Deck.id)))
    total = count_result.scalar() or 0

    return DeckListResponse(decks=list(decks), total=total)


@router.get("/decks/{deck_id}", response_model=DeckResponse)
async def get_deck_by_id(
    deck_id: int,
    db: AsyncSession = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Get a specific deck."""
    result = await db.execute(select(Deck).where(Deck.id == deck_id))
    deck = result.scalar_one_or_none()

    if not deck:
        raise HTTPException(status_code=404, detail="Deck not found")
    return deck
```
